# DAST bằng OWASP ZAP — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Thêm OWASP ZAP làm nguồn finding động bên cạnh SAST, để chuỗi đề xuất → phê duyệt → probe có bằng chứng runtime thay vì luôn trả `inconclusive`.

**Architecture:** ZAP chạy như service Docker thứ năm trong `sentinel-net`, **không bind cổng host nào**, điều khiển bằng `scripts/scan-zap.sh` theo đúng khuôn `scripts/scan-opengrep.sh` đã có. Nó quét WebGoat trực tiếp (không qua Gateway) trong một session đã đăng nhập. Output đi qua ranh giới D (`scan_injection` → `redact` → `wrap_untrusted`) rồi mới vào `findings.json`. Một module đối chiếu tất định nối finding SAST với endpoint ZAP chạm được, và kết quả đó ghi đè trường `reachability` mà Agent tự khai.

**Tech Stack:** Python 3.10+, Docker Compose, OWASP ZAP (`zaproxy/zap-stable`), pytest, jq, bash.

**Spec:** [`docs/superpowers/specs/2026-08-22-dast-zap-design.md`](../specs/2026-08-22-dast-zap-design.md)

`infra/docker/zap/sentinel_hook.py`. Điều chỉnh tên hàm và chữ ký theo Step 1–2:

```python
"""Hook cho zap-baseline.py: đăng nhập WebGoat, giữ session, dump endpoint.

Chạy BÊN TRONG container ZAP. Sentinel không nói chuyện với ZAP API từ host —
đó là lý do ZAP không cần bind cổng nào.

WebGoat bắt đăng nhập cho mọi lesson. Không có session thì spider chỉ thấy
trang login và scan trở nên vô nghĩa.
"""

import http.cookiejar
import json
import os
import urllib.parse
import urllib.request
import uuid

TARGET = os.environ.get("SENTINEL_ZAP_TARGET", "http://webgoat:8080/WebGoat")
ENDPOINTS_OUT = os.environ.get(
    "SENTINEL_ZAP_ENDPOINTS_OUT", "/zap/wrk/artifacts/zap-endpoints.json"
)

# UserForm.java:22-34 — username [a-z0-9-]* dài 6-45, password dài 6-10.
_USERNAME = "sentinel-" + uuid.uuid4().hex[:8]
_PASSWORD = "sent1nel"


def _post_form(url: str, fields: dict) -> str | None:
    """POST form, trả JSESSIONID nếu server phát ra, None nếu không."""
    jar = http.cookiejar.CookieJar()
    opener = urllib.request.build_opener(urllib.request.HTTPCookieProcessor(jar))
    body = urllib.parse.urlencode(fields).encode()
    request = urllib.request.Request(url, data=body, method="POST")
    request.add_header("Content-Type", "application/x-www-form-urlencoded")
    try:
        with opener.open(request, timeout=30) as response:
            response.read()
    except OSError as exc:
        print(f"[sentinel] POST {url} hỏng: {exc}")
        return None
    return next((c.value for c in jar if c.name == "JSESSIONID"), None)


def _acquire_session() -> str | None:
    """Đăng ký tài khoản mới; RegistrationController.java:60 tự login luôn."""
    session = _post_form(
        f"{TARGET}/register.mvc",
        {
            "username": _USERNAME,
            "password": _PASSWORD,
            "matchingPassword": _PASSWORD,
            "agree": "agree",
        },
    )
    if session:
        print(f"[sentinel] Có session qua đăng ký, user={_USERNAME}")
        return session
    # Dự phòng: WebSecurityConfig.java:50-54 — formLogin param username/password.
    session = _post_form(
        f"{TARGET}/login", {"username": _USERNAME, "password": _PASSWORD}
    )
    if session:
        print("[sentinel] Có session qua login")
    else:
        print("[sentinel] KHÔNG lấy được session — scan sẽ chỉ thấy trang công khai")
    return session


def zap_started(zap, target):
    """Nạp session vào ZAP và chặn spider tự đăng xuất."""
    session = _acquire_session()
    if session:
        zap.replacer.add_rule(
            description="sentinel-session",
            enabled=True,
            matchtype="REQ_HEADER",
            matchregex=False,
            matchstring="Cookie",
            replacement=f"JSESSIONID={session}",
        )
    # Thiếu dòng này thì spider tự bấm logout giữa chừng và mọi request sau
    # đó chạy ẩn danh — scan vẫn "thành công" nhưng rỗng.
    zap.spider.exclude_from_scan(".*logout.*")
    return zap, target


def zap_pre_shutdown(zap):
    """Dump bản đồ endpoint — đây là input của đối chiếu và của slice C."""
    endpoints = []
    seen = set()
    for message in zap.core.messages(baseurl=TARGET):
        header = message.get("requestHeader", "")
        first_line = header.split("\r\n", 1)[0]
        parts = first_line.split(" ")
        if len(parts) < 2:
            continue
        method, url = parts[0], parts[1]
        split = urllib.parse.urlsplit(url)
        params = sorted(urllib.parse.parse_qs(split.query).keys())
        key = (method, split.path, tuple(params))
        if key in seen:
            continue
        seen.add(key)
        endpoints.append({"method": method, "url": url, "params": params})

    payload = {"schema_version": "1.0", "target": TARGET, "endpoints": endpoints}
    with open(ENDPOINTS_OUT, "w", encoding="utf-8") as handle:
        json.dump(payload, handle, ensure_ascii=False, indent=2)
    print(f"[sentinel] Ghi {len(endpoints)} endpoint vào {ENDPOINTS_OUT}")
    return zap
```

```python
"""Chuẩn hoá report OWASP ZAP về schema finding dùng chung.

Khác `normalizer.py` (OpenGrep) ở hai chỗ. Thứ nhất, vị trí là URL chứ không
phải file:line — schema chung đã lường trước bằng tên trường `file_or_url`.
Thứ hai, một alert ZAP trải trên nhiều URL: WebGoat gọi `headers.disable()`
(WebSecurityConfig.java:62) nên alert thiếu security header bắn trên MỌI URL.
Một-finding-một-URL sẽ cho ra hàng nghìn finding và đốt token ở bước analyze.
Nên ta gộp theo loại alert và giữ danh sách instance đã cắt bớt.
"""

from __future__ import annotations

from typing import Any

# ZAP dùng riskcode 0..3; riskdesc là chuỗi "High (Medium)".
ZAP_RISK_TO_SEVERITY: dict[str, str] = {
    "3": "high",
    "2": "medium",
    "1": "low",
    "0": "info",
}

DEFAULT_MAX_INSTANCES = 20


def _severity_of(alert: dict[str, Any]) -> str:
    riskcode = str(alert.get("riskcode", "")).strip()
    if riskcode in ZAP_RISK_TO_SEVERITY:
        return ZAP_RISK_TO_SEVERITY[riskcode]
    head = str(alert.get("riskdesc", "")).split(" ", 1)[0].strip().lower()
    return {"high": "high", "medium": "medium", "low": "low",
            "informational": "info"}.get(head, "low")


def _first_nonempty(*values: Any) -> str | None:
    for value in values:
        if isinstance(value, str) and value.strip():
            return value.strip()
    return None


def _instances_of(alert: dict[str, Any], limit: int) -> tuple[list[dict], int]:
    raw = alert.get("instances")
    items = raw if isinstance(raw, list) else []
    collected = [
        {
            "url": str(item.get("uri") or ""),
            "method": str(item.get("method") or "GET").upper(),
            "param": str(item.get("param") or "") or None,
            "evidence": str(item.get("evidence") or "") or None,
            "attack": str(item.get("attack") or "") or None,
        }
        for item in items
        if isinstance(item, dict)
    ]
    return collected[:limit], len(collected)


def normalize_zap_report(
    raw: dict[str, Any], *, max_instances: int = DEFAULT_MAX_INSTANCES
) -> list[dict[str, Any]]:
    """Gộp alert ZAP theo pluginid, trả danh sách finding schema chung."""
    if not isinstance(raw, dict):
        raise ValueError("Report ZAP phải là JSON object")
    sites = raw.get("site")
    if not isinstance(sites, list):
        raise ValueError("Report ZAP thiếu mảng site")

    # pluginid -> alert đã gộp instance từ mọi site.
    merged: dict[str, dict[str, Any]] = {}
    for site in sites:
        if not isinstance(site, dict):
            continue
        for alert in site.get("alerts") or []:
            if not isinstance(alert, dict):
                continue
            key = str(alert.get("pluginid") or alert.get("alertRef") or alert.get("name"))
            if key not in merged:
                merged[key] = dict(alert)
                merged[key]["instances"] = list(alert.get("instances") or [])
            else:
                merged[key]["instances"].extend(alert.get("instances") or [])

    findings: list[dict[str, Any]] = []
    for index, (plugin_id, alert) in enumerate(sorted(merged.items()), start=1):
        instances, total = _instances_of(alert, max_instances)
        primary_url = instances[0]["url"] if instances else ""
        findings.append(
            {
                "id": f"zap-{index:03d}",
                "tool": "zap",
                "tool_version": str(raw.get("@version") or "unknown"),
                "severity": _severity_of(alert),
                "file_or_url": primary_url,
                "line": None,
                "title": _first_nonempty(alert.get("name")) or f"ZAP {plugin_id}",
                "rule_id": str(plugin_id),
                "cwe": _first_nonempty(alert.get("cweid")),
                "owasp": None,
                "message": _first_nonempty(alert.get("desc")) or "",
                "confidence": _first_nonempty(alert.get("confidence")),
                "fingerprint": f"zap:{plugin_id}",
                "raw_check_id": str(plugin_id),
                "instances": instances,
                "instances_total": total,
            }
        )
    return findings


__all__ = ["ZAP_RISK_TO_SEVERITY", "DEFAULT_MAX_INSTANCES", "normalize_zap_report"]
```

```python
import copy
from pathlib import Path

from project_sentinel.guardrails.events import append_event
from project_sentinel.guardrails.injection import scan as scan_injection
from project_sentinel.guardrails.injection import wrap_untrusted
from project_sentinel.guardrails.redaction import redact

# Hai trường này mang nguyên văn response của ứng dụng đích.
_UNTRUSTED_FIELDS = ("evidence", "attack")


def _clean_one(
    text: str, events_path: Path | None, run_id: str, counts: dict[str, int]
) -> str:
    """Quét injection TRƯỚC rồi mới che.

    Thứ tự này có chủ ý và giống hệt đường probe: bỏ chỉ dẫn tấn công đi trước
    thì bộ che không phải xử lý nội dung do kẻ tấn công dựng riêng để đánh lừa
    chính nó.
    """
    verdict = scan_injection(text)
    if verdict.verdict != "clean":
        counts["injection"] = counts.get("injection", 0) + 1
        if events_path is not None:
            append_event(
                events_path,
                run_id=run_id,
                kind="injection",
                detail={
                    "source": "zap_alert",
                    "patterns": sorted({m.pattern_name for m in verdict.matches}),
                },
            )
    redacted, events = redact(verdict.sanitized_text)
    if events:
        counts["redaction"] = counts.get("redaction", 0) + sum(e.count for e in events)
    return wrap_untrusted(redacted)


def sanitize_zap_findings(
    findings: list[dict[str, Any]],
    *,
    events_path: str | Path | None = None,
    run_id: str = "",
) -> tuple[list[dict[str, Any]], dict[str, int]]:
    """Áp ranh giới D lên mọi chuỗi ZAP mang nội dung của ứng dụng đích.

    Trả về (findings đã sạch, số đếm). Không sửa input tại chỗ.
    """
    path = Path(events_path) if events_path is not None else None
    counts: dict[str, int] = {"injection": 0, "redaction": 0}
    cleaned = copy.deepcopy(findings)
    for finding in cleaned:
        for instance in finding.get("instances") or []:
            for field in _UNTRUSTED_FIELDS:
                value = instance.get(field)
                if isinstance(value, str) and value:
                    instance[field] = _clean_one(value, path, run_id, counts)
    return cleaned, counts
```

```python
def _merge_zap_into_findings(root: Path, *, run_id: str) -> int:
    """Trộn finding ZAP vào findings.json. Trả về số finding đã thêm.

    Không có report ZAP thì đây là no-op — DAST là tuỳ chọn, và một run chỉ
    SAST phải cho ra đúng findings.json như trước khi có task này.
    """
    alerts_path = root / "zap-alerts.json"
    findings_path = root / "findings.json"
    if not alerts_path.exists() or not findings_path.exists():
        return 0

    try:
        raw = json.loads(alerts_path.read_text(encoding="utf-8"))
        zap_findings = normalize_zap_report(raw)
    except (OSError, ValueError, json.JSONDecodeError) as exc:
        raise StepFailure(f"Không đọc được zap-alerts.json: {exc}") from exc

    zap_findings, _counts = sanitize_zap_findings(
        zap_findings, events_path=root / "events.jsonl", run_id=run_id
    )

    payload = json.loads(findings_path.read_text(encoding="utf-8"))
    existing = payload.get("findings") or []
    payload["findings"] = [*existing, *zap_findings]
    payload["count"] = len(payload["findings"])
    payload["sources"] = {"opengrep": len(existing), "zap": len(zap_findings)}
    findings_path.write_text(
        json.dumps(payload, indent=2, ensure_ascii=False) + "\n", encoding="utf-8"
    )
    return len(zap_findings)
```

```python
def _dast_evidence(finding: dict) -> SourceEvidence:
    """Bằng chứng của một finding động là chính alert đã qua ranh giới D.

    Nội dung `evidence` ở đây đã được scan injection, che và bọc untrusted từ
    bước normalize. Hàm này không làm sạch lại — nó chỉ trình bày.
    """
    instances = finding.get("instances") or []
    if not instances:
        return SourceEvidence(
            path=str(finding.get("file_or_url") or ""),
            start_line=0,
            end_line=0,
            content="",
            error="Finding DAST không có instance nào để làm bằng chứng",
        )

    total = finding.get("instances_total", len(instances))
    lines = [f"Alert: {finding.get('title') or finding.get('rule_id')}",
             f"Số vị trí bị ảnh hưởng: {total}"]
    for instance in instances:
        lines.append(
            f"- {instance.get('method', 'GET')} {instance.get('url', '')}"
            + (f" [param={instance['param']}]" if instance.get("param") else "")
        )
        if instance.get("evidence"):
            lines.append(f"  {instance['evidence']}")

    return SourceEvidence(
        path=str(instances[0].get("url") or finding.get("file_or_url") or ""),
        start_line=0,
        end_line=0,
        content="\n".join(lines),
    )


def evidence_for_finding(
    finding: dict,
    *,
    project_root: Path,
    target_root: Path,
    radius: int = 4,
) -> SourceEvidence:
    """Chọn đường trích bằng chứng theo hình dạng của finding.

    Finding có file+line đi đúng đường cũ, không đổi một byte hành vi
    (AGENTS.md §2.1). Chỉ finding không có vị trí source mới rẽ sang DAST.
    """
    location = finding.get("file_or_url")
    line = finding.get("line")
    if isinstance(line, int) and line > 0 and location and "://" not in str(location):
        return extract_source_window(
            project_root, target_root, str(location), line, radius
        )
    if str(finding.get("tool")) == "zap":
        return _dast_evidence(finding)
    return extract_source_window(
        project_root, target_root, str(location or ""), line or 0, radius
    )
```

```python
"""Nối finding tĩnh với endpoint runtime mà ZAP thật sự chạm tới.

Vì sao cần. Finding SAST là `SqlInjectionLesson5a.java:47`; endpoint DAST là
`/WebGoat/SqlInjection/attack5a`. Không có gì nối hai thứ đó một cách hiển
nhiên. Cầu nối là annotation route trong chính file chứa finding — trích ra
được bằng cách đọc file, tất định, không hỏi LLM.

Vì tất định nên kết quả của nó đủ tin cậy để GHI ĐÈ trường `reachability` mà
Agent tự khai. Xem `calibration.py`.
"""

from __future__ import annotations

import re
from pathlib import Path

_CLASS_MAPPING = re.compile(r'@RequestMapping\s*\(\s*(?:value\s*=\s*)?"([^"]+)"')
_METHOD_MAPPING = re.compile(
    r'@(?:Get|Post|Put|Delete|Patch|Request)Mapping\s*\(\s*(?:value\s*=\s*)?"([^"]+)"'
)
_MAX_BYTES = 512 * 1024


def _join(prefix: str | None, suffix: str) -> str:
    if not prefix:
        return "/" + suffix.lstrip("/")
    return "/" + prefix.strip("/") + "/" + suffix.lstrip("/")


def extract_route(source_path: Path) -> str | None:
    """Trả route Spring khai trong file, hoặc None nếu không có.

    Đọc có giới hạn kích thước: file khổng lồ không được làm treo bước
    normalize. Không đọc được thì trả None — thiếu bằng chứng, không phải lỗi.
    """
    try:
        if not source_path.is_file() or source_path.stat().st_size > _MAX_BYTES:
            return None
        text = source_path.read_text(encoding="utf-8", errors="replace")
    except OSError:
        return None

    class_match = _CLASS_MAPPING.search(text)
    class_prefix = class_match.group(1) if class_match else None

    for match in _METHOD_MAPPING.finditer(text):
        # Bỏ qua chính cái @RequestMapping mức class đã bắt ở trên.
        if class_match and match.start() == class_match.start():
            continue
        return _join(class_prefix, match.group(1))

    return _join(None, class_prefix) if class_prefix else None


__all__ = ["extract_route"]
```

```python
import urllib.parse
from typing import Any

STRENGTHS: tuple[str, ...] = (
    "no_route",
    "route_known_not_reached",
    "reachable",
    "reachable_and_alerted",
)


def _paths_of(endpoints: dict[str, Any]) -> set[str]:
    paths = set()
    for item in endpoints.get("endpoints") or []:
        url = str(item.get("url") or "")
        if url:
            paths.add(urllib.parse.urlsplit(url).path)
    return paths


def _route_matches(route: str, observed_paths: set[str]) -> str | None:
    """Route khai trong code là tương đối với context path (/WebGoat)."""
    suffix = "/" + route.strip("/")
    for path in observed_paths:
        if path == suffix or path.endswith(suffix):
            return path
    return None


def correlate(
    findings: list[dict[str, Any]],
    endpoints: dict[str, Any],
    *,
    project_root: Path,
) -> list[dict[str, Any]]:
    """Gắn khối `runtime_evidence` vào mỗi finding TĨNH. Không sửa input."""
    observed = _paths_of(endpoints)

    # path runtime -> các finding ZAP có alert trên chính path đó
    alerts_by_path: dict[str, list[str]] = {}
    for finding in findings:
        if str(finding.get("tool")) != "zap":
            continue
        for instance in finding.get("instances") or []:
            path = urllib.parse.urlsplit(str(instance.get("url") or "")).path
            if path:
                alerts_by_path.setdefault(path, []).append(str(finding["id"]))

    result: list[dict[str, Any]] = []
    for finding in findings:
        item = dict(finding)
        if str(finding.get("tool")) == "zap":
            # Finding động ĐÃ là bằng chứng runtime.
            result.append(item)
            continue

        relative = str(finding.get("file_or_url") or "")
        route = extract_route(project_root / relative) if relative else None

        if not route:
            item["runtime_evidence"] = {
                "route": None, "route_source": None, "observed": None,
                "strength": "no_route", "dast_alerts": [],
            }
            result.append(item)
            continue

        matched = _route_matches(route, observed)
        alerts = sorted(set(alerts_by_path.get(matched, []))) if matched else []
        if matched and alerts:
            strength = "reachable_and_alerted"
        elif matched:
            strength = "reachable"
        else:
            strength = "route_known_not_reached"

        item["runtime_evidence"] = {
            "route": route,
            "route_source": relative,
            "observed": matched,
            "strength": strength,
            "dast_alerts": alerts,
        }
        result.append(item)
    return result


__all__ = ["STRENGTHS", "correlate", "extract_route"]
```

```python
"""Đề xuất ỨNG VIÊN allowlist từ endpoint ZAP thật sự chạm tới.

Hai kỷ luật, cả hai đều cố ý.

1. File này sinh ra ỨNG VIÊN, không phải allowlist. Người review và tự tay
   chuyển sang `configs/gateway/endpoint-allowlist.json`. Để một tool tự nới
   bề mặt tấn công là bỏ mất cổng người duyệt.
2. Không sinh allowlist Nginx. Hai allowlist chỉ có giá trị khi được suy ra
   ĐỘC LẬP; sinh cả hai từ cùng một nguồn làm lớp phòng thủ kép sập còn một.
"""

from __future__ import annotations

import urllib.parse
from typing import Any

# WebSecurityConfig.java:34-44 — bề mặt truy cập được khi CHƯA đăng nhập.
PERMIT_ALL_PREFIXES: tuple[str, ...] = (
    "/favicon.ico", "/css/", "/images/", "/js/", "/fonts/", "/plugins/",
    "/registration", "/register.mvc", "/actuator/", "/login",
)


def _is_unauthenticated_reachable(path: str) -> bool:
    tail = path.split("/WebGoat", 1)[-1] if "/WebGoat" in path else path
    return any(tail.startswith(prefix) for prefix in PERMIT_ALL_PREFIXES)


def build_candidates(
    endpoints: dict[str, Any], allowlist: dict[str, Any]
) -> dict[str, Any]:
    """Trả về ứng viên allowlist kèm provenance runtime và cờ trung thực."""
    known = {
        str(item.get("path"))
        for item in allowlist.get("endpoints") or []
        if item.get("path")
    }

    candidates: dict[str, dict[str, Any]] = {}
    for item in endpoints.get("endpoints") or []:
        url = str(item.get("url") or "")
        if not url:
            continue
        path = urllib.parse.urlsplit(url).path
        if not path or path in known or path in candidates:
            continue
        method = str(item.get("method") or "GET").upper()
        candidates[path] = {
            "path": path,
            # GET-only, không payload. POST và payload phải do người viết tay.
            "allowed_methods": ["GET"],
            "payload_kind": None,
            "observed_method": method,
            "observed_params": list(item.get("params") or []),
            "unauthenticated_reachable": _is_unauthenticated_reachable(path),
            "source": f"ZAP quan sát {method} {url}",
        }

    return {
        "schema_version": "1.0",
        # Khoá tên "candidates" chứ KHÔNG phải "endpoints": file này không bao
        # giờ được parse nhầm như một allowlist thật.
        "candidates": sorted(candidates.values(), key=lambda c: c["path"]),
    }


__all__ = ["PERMIT_ALL_PREFIXES", "build_candidates"]
```
